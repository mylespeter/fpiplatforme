'use client';

import { useState, useEffect } from 'react';
import { useAuth } from '@/context/AuthContext';
import { supabase } from '@/lib/supabase';
import { 
  FileText, Clock, CheckCircle, AlertCircle, 
  TrendingUp, Users, Search, ArrowUpRight,
  Loader2, Briefcase, BarChart3, PieChart,
  Activity, Target, Shield
} from 'lucide-react';
import Link from 'next/link';

// Types
type ProjetStats = {
  id: number;
  nom_projet: string;
  promoteur_nom_complet: string;
  montant_sollicite: number;
  etape: string;
  created_at: string;
  technicien_id: number | null;
  decision?: string;
};

// Composant Donut Chart
function DonutChart({ 
  segments, 
  total, 
  size = 160,
  strokeWidth = 25 
}: { 
  segments: { value: number; color: string; label: string }[];
  total: number;
  size?: number;
  strokeWidth?: number;
}) {
  const radius = (size - strokeWidth) / 2;
  const circumference = 2 * Math.PI * radius;
  const center = size / 2;

  let currentAngle = -90;

  return (
    <div className="relative" style={{ width: size, height: size }}>
      <svg width={size} height={size} className="transform -rotate-90">
        {total === 0 ? (
          <circle
            cx={center}
            cy={center}
            r={radius}
            fill="none"
            stroke="#e5e7eb"
            strokeWidth={strokeWidth}
          />
        ) : (
          segments.map((segment, index) => {
            const percentage = (segment.value / total) * 100;
            const dashArray = (percentage * circumference) / 100;
            const angle = (segment.value / total) * 360;
            
            const startX = center + radius * Math.cos((currentAngle * Math.PI) / 180);
            const startY = center + radius * Math.sin((currentAngle * Math.PI) / 180);
            
            currentAngle += angle;
            
            const endX = center + radius * Math.cos((currentAngle * Math.PI) / 180);
            const endY = center + radius * Math.sin((currentAngle * Math.PI) / 180);
            
            const largeArc = angle > 180 ? 1 : 0;

            return (
              <g key={index}>
                <path
                  d={`M ${startX} ${startY} A ${radius} ${radius} 0 ${largeArc} 1 ${endX} ${endY}`}
                  fill="none"
                  stroke={segment.color}
                  strokeWidth={strokeWidth}
                  strokeLinecap="round"
                  className="transition-all duration-500"
                />
              </g>
            );
          })
        )}
      </svg>
      
      {/* Centre */}
      <div className="absolute inset-0 flex flex-col items-center justify-center">
        <span className="text-2xl font-bold text-gray-900">{total}</span>
        <span className="text-xs text-gray-500">Total</span>
      </div>
    </div>
  );
}

export default function DashboardTechnique() {
  const { user } = useAuth();
  const [loading, setLoading] = useState(true);
  const [projets, setProjets] = useState<ProjetStats[]>([]);
  const [searchTerm, setSearchTerm] = useState('');

  useEffect(() => {
    chargerProjets();
  }, []);

  const chargerProjets = async () => {
    try {
      const { data: projetsData, error } = await supabase
        .from('projets_fpi')
        .select(`
          id,
          nom_projet,
          promoteur_nom_complet,
          montant_sollicite,
          etape,
          created_at
        `)
        .order('created_at', { ascending: false });

      if (error) throw error;

      // Récupérer les techniciens assignés et décisions
      const projetIds = (projetsData || []).map(p => p.id);
      
      const { data: rapports } = await supabase
        .from('rapport_analyse')
        .select('projet_id, technicien_id, decision')
        .in('projet_id', projetIds);

      const projetsAvecTechnicien = (projetsData || []).map(projet => {
        const rapport = (rapports || []).find(r => r.projet_id === projet.id);
        return {
          ...projet,
          technicien_id: rapport?.technicien_id || null,
          decision: rapport?.decision || null
        };
      });

      setProjets(projetsAvecTechnicien);
    } catch (error) {
      console.error('Erreur chargement projets:', error);
    } finally {
      setLoading(false);
    }
  };

  // Statistiques
  const stats = {
    total: projets.length,
    aAnalyser: projets.filter(p => p.etape === 'soumis' || (p.etape === 'analyse_tech' && !p.technicien_id)).length,
    enCours: projets.filter(p => p.etape === 'analyse_tech' && p.technicien_id).length,
    analyses: projets.filter(p => p.etape === 'comité_crédit').length,
    approuves: projets.filter(p => p.etape === 'financement_approuve').length,
    rejetes: projets.filter(p => p.etape === 'financement_rejete').length,
  };

  // Statistiques des décisions
  const statsDecisions = {
    favorables: projets.filter(p => p.decision === 'favorable').length,
    defavorables: projets.filter(p => p.decision === 'defavorable').length,
    reserves: projets.filter(p => p.decision === 'reserve').length,
    enAttente: projets.filter(p => p.etape === 'analyse_tech' && !p.decision).length,
  };

  // Projets récents
  const projetsRecents = projets
    .filter(p => 
      searchTerm === '' || 
      p.nom_projet.toLowerCase().includes(searchTerm.toLowerCase()) ||
      p.promoteur_nom_complet.toLowerCase().includes(searchTerm.toLowerCase())
    )
    .slice(0, 5);

  const formatMontant = (m: number) => 
    new Intl.NumberFormat('fr-FR', { 
      style: 'currency', 
      currency: 'USD', 
      maximumFractionDigits: 0 
    }).format(m);

  const formatDate = (d: string) => 
    new Date(d).toLocaleDateString('fr-FR', { 
      day: 'numeric', 
      month: 'short', 
      year: 'numeric' 
    });

  const getEtapeBadge = (etape: string) => {
    const badges: Record<string, { label: string; color: string }> = {
      'soumis': { label: 'Soumis', color: 'bg-blue-100 text-blue-700' },
      'analyse_tech': { label: 'En analyse', color: 'bg-amber-100 text-amber-700' },
      'comité_crédit': { label: 'Analysé', color: 'bg-purple-100 text-purple-700' },
      'financement_approuve': { label: 'Approuvé', color: 'bg-green-100 text-green-700' },
      'financement_rejete': { label: 'Rejeté', color: 'bg-red-100 text-red-700' },
    };
    return badges[etape] || { label: etape, color: 'bg-gray-100 text-gray-700' };
  };

  // Segments pour le donut des étapes
  const segmentsEtapes = [
    { value: stats.aAnalyser, color: '#F59E0B', label: 'À analyser' },
    { value: stats.enCours, color: '#8B5CF6', label: 'En cours' },
    { value: stats.analyses, color: '#3B82F6', label: 'Analysés' },
    { value: stats.approuves, color: '#10B981', label: 'Approuvés' },
    { value: stats.rejetes, color: '#EF4444', label: 'Rejetés' },
  ];

  // Segments pour le donut des décisions
  const segmentsDecisions = [
    { value: statsDecisions.favorables, color: '#10B981', label: 'Favorable' },
    { value: statsDecisions.defavorables, color: '#EF4444', label: 'Défavorable' },
    { value: statsDecisions.reserves, color: '#F59E0B', label: 'Réservé' },
    { value: statsDecisions.enAttente, color: '#6B7280', label: 'En attente' },
  ];

  if (loading) {
    return (
      <div className="flex items-center justify-center min-h-[60vh]">
        <Loader2 className="h-8 w-8 animate-spin text-primary" />
      </div>
    );
  }

  return (
    <div className="space-y-6">
      {/* En-tête */}
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-2xl font-bold text-gray-900">
            Tableau de bord technique
          </h1>
          <p className="text-sm text-gray-500 mt-1">
            Bienvenue, {user?.username} • Vue d'ensemble des analyses
          </p>
        </div>
        <Link
          href="/dashboard/technique/projets"
          className="flex items-center gap-2 px-4 py-2 text-sm font-medium text-white bg-primary rounded-lg hover:bg-primary-dark transition-colors"
        >
          <Briefcase className="h-4 w-4" />
          Voir tous les projets
        </Link>
      </div>

      {/* Cartes statistiques rapides */}
      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
        <div className="bg-white rounded-xl p-4 border border-gray-200">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 bg-amber-100 rounded-lg flex items-center justify-center">
              <Clock className="h-5 w-5 text-amber-600" />
            </div>
            <div>
              <p className="text-xs text-gray-500">À analyser</p>
              <p className="text-xl font-bold text-gray-900">{stats.aAnalyser}</p>
            </div>
          </div>
        </div>

        <div className="bg-white rounded-xl p-4 border border-gray-200">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 bg-purple-100 rounded-lg flex items-center justify-center">
              <Activity className="h-5 w-5 text-purple-600" />
            </div>
            <div>
              <p className="text-xs text-gray-500">En cours</p>
              <p className="text-xl font-bold text-gray-900">{stats.enCours}</p>
            </div>
          </div>
        </div>

        <div className="bg-white rounded-xl p-4 border border-gray-200">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 bg-blue-100 rounded-lg flex items-center justify-center">
              <CheckCircle className="h-5 w-5 text-blue-600" />
            </div>
            <div>
              <p className="text-xs text-gray-500">Analysés</p>
              <p className="text-xl font-bold text-gray-900">{stats.analyses}</p>
            </div>
          </div>
        </div>

        <div className="bg-white rounded-xl p-4 border border-gray-200">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 bg-green-100 rounded-lg flex items-center justify-center">
              <Target className="h-5 w-5 text-green-600" />
            </div>
            <div>
              <p className="text-xs text-gray-500">Approuvés</p>
              <p className="text-xl font-bold text-gray-900">{stats.approuves}</p>
            </div>
          </div>
        </div>
      </div>

      {/* Graphiques Donut */}
      <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
        {/* Donut : Statut des projets */}
        <div className="bg-white rounded-xl border border-gray-200 p-6">
          <div className="flex items-center gap-2 mb-6">
            <PieChart className="h-5 w-5 text-gray-400" />
            <h2 className="text-lg font-semibold text-gray-900">
              Répartition des projets
            </h2>
          </div>
          
          <div className="flex items-center justify-center">
            <DonutChart 
              segments={segmentsEtapes} 
              total={stats.total}
              size={200}
              strokeWidth={30}
            />
          </div>

          {/* Légende */}
          <div className="mt-6 space-y-2">
            {segmentsEtapes.map((segment, index) => (
              <div key={index} className="flex items-center justify-between text-sm">
                <div className="flex items-center gap-2">
                  <div 
                    className="w-3 h-3 rounded-full" 
                    style={{ backgroundColor: segment.color }}
                  />
                  <span className="text-gray-600">{segment.label}</span>
                </div>
                <div className="flex items-center gap-2">
                  <span className="font-semibold text-gray-900">{segment.value}</span>
                  <span className="text-xs text-gray-400">
                    ({stats.total > 0 ? Math.round((segment.value / stats.total) * 100) : 0}%)
                  </span>
                </div>
              </div>
            ))}
          </div>
        </div>

        {/* Donut : Décisions */}
        <div className="bg-white rounded-xl border border-gray-200 p-6">
          <div className="flex items-center gap-2 mb-6">
            <Shield className="h-5 w-5 text-gray-400" />
            <h2 className="text-lg font-semibold text-gray-900">
              Décisions d'analyse
            </h2>
          </div>
          
          <div className="flex items-center justify-center">
            <DonutChart 
              segments={segmentsDecisions} 
              total={statsDecisions.favorables + statsDecisions.defavorables + statsDecisions.reserves + statsDecisions.enAttente}
              size={200}
              strokeWidth={30}
            />
          </div>

          {/* Légende */}
          <div className="mt-6 space-y-2">
            {segmentsDecisions.map((segment, index) => (
              <div key={index} className="flex items-center justify-between text-sm">
                <div className="flex items-center gap-2">
                  <div 
                    className="w-3 h-3 rounded-full" 
                    style={{ backgroundColor: segment.color }}
                  />
                  <span className="text-gray-600">{segment.label}</span>
                </div>
                <div className="flex items-center gap-2">
                  <span className="font-semibold text-gray-900">{segment.value}</span>
                  <span className="text-xs text-gray-400">
                    ({stats.total > 0 ? Math.round((segment.value / stats.total) * 100) : 0}%)
                  </span>
                </div>
              </div>
            ))}
          </div>
        </div>
      </div>

      {/* Projets récents */}
      <div className="bg-white rounded-xl border border-gray-200">
        <div className="p-6 border-b border-gray-200">
          <div className="flex items-center justify-between">
            <h2 className="text-lg font-semibold text-gray-900">
              Projets récents
            </h2>
            <div className="relative">
              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
              <input
                type="text"
                placeholder="Rechercher un projet..."
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
                className="pl-10 pr-4 py-2 text-sm border border-gray-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-primary/20 w-64"
              />
            </div>
          </div>
        </div>
        
        <div className="divide-y divide-gray-100">
          {projetsRecents.length === 0 ? (
            <div className="p-12 text-center">
              <FileText className="h-12 w-12 mx-auto mb-3 text-gray-300" />
              <p className="text-gray-500">Aucun projet trouvé</p>
            </div>
          ) : (
            projetsRecents.map((projet) => {
              const badge = getEtapeBadge(projet.etape);
              return (
                <Link
                  key={projet.id}
                  href={`/dashboard/technique/projets/${projet.id}`}
                  className="flex items-center justify-between p-4 hover:bg-gray-50 transition-colors"
                >
                  <div className="flex-1 min-w-0">
                    <div className="flex items-center gap-3">
                      <h3 className="text-sm font-medium text-gray-900 truncate">
                        {projet.nom_projet}
                      </h3>
                      <span className={`px-2 py-0.5 rounded-full text-xs font-medium ${badge.color}`}>
                        {badge.label}
                      </span>
                      {projet.decision && (
                        <span className={`px-2 py-0.5 rounded-full text-xs font-medium ${
                          projet.decision === 'favorable' ? 'bg-green-100 text-green-700' :
                          projet.decision === 'defavorable' ? 'bg-red-100 text-red-700' :
                          'bg-yellow-100 text-yellow-700'
                        }`}>
                          {projet.decision}
                        </span>
                      )}
                    </div>
                    <p className="text-xs text-gray-500 mt-1">
                      {projet.promoteur_nom_complet} • {formatDate(projet.created_at)}
                    </p>
                  </div>
                  
                  <div className="flex items-center gap-4 ml-4">
                    <p className="text-sm font-semibold text-gray-900">
                      {formatMontant(projet.montant_sollicite || 0)}
                    </p>
                    <ArrowUpRight className="h-4 w-4 text-gray-400" />
                  </div>
                </Link>
              );
            })
          )}
        </div>

        {projets.length > 5 && (
          <div className="p-4 border-t border-gray-100 bg-gray-50/50">
            <Link
              href="/dashboard/technique/projets"
              className="flex items-center justify-center gap-2 w-full py-2 text-sm text-primary hover:bg-white rounded-lg transition-colors"
            >
              Voir tous les projets ({projets.length})
              <ArrowUpRight className="h-4 w-4" />
            </Link>
          </div>
        )}
      </div>
    </div>
  );
}
